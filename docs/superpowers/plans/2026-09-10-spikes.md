**Spec:** `docs/superpowers/specs/2026-09-09-live-translation-design.md`

## Global Constraints

- Gemini model id is exactly `gemini-3.5-live-translate-preview`.
- Live API audio input: 16-bit PCM, mono, 16 kHz. Output: 16-bit PCM, 24 kHz.
- Attendees are served over plain HTTP. `RTCPeerConnection`, WebCodecs `AudioDecoder`, `AudioWorklet` and Wake Lock are unavailable and must not appear in any spike conclusion as an option.
- `GEMINI_API_KEY` is read from the environment. Never hardcode it, never log it.
- All spike code lives under `spikes/` and is excluded from the pnpm workspace.
- Findings go in `docs/superpowers/spike-findings.md`. Every task appends a section with a dated **Decision** line.

`spikes/shared/pcm.ts`:
```ts
import { readFile } from "node:fs/promises";

export function chunkPcm(pcm: Buffer, bytesPerChunk: number): Buffer[] {
  const out: Buffer[] = [];
  for (let i = 0; i + bytesPerChunk <= pcm.length; i += bytesPerChunk) {
    out.push(pcm.subarray(i, i + bytesPerChunk));
  }
  return out;
}

export async function readWavAsPcm16(
  path: string,
): Promise<{ pcm: Buffer; sampleRate: number; channels: number }> {
  const buf = await readFile(path);
  if (buf.toString("ascii", 0, 4) !== "RIFF" || buf.toString("ascii", 8, 12) !== "WAVE") {
    throw new Error(`not a RIFF/WAVE file: ${path}`);
  }

  let sampleRate = 0;
  let channels = 0;
  let bitsPerSample = 0;
  let offset = 12;

  while (offset + 8 <= buf.length) {
    const id = buf.toString("ascii", offset, offset + 4);
    const size = buf.readUInt32LE(offset + 4);
    const body = offset + 8;

    if (id === "fmt ") {
      channels = buf.readUInt16LE(body + 2);
      sampleRate = buf.readUInt32LE(body + 4);
      bitsPerSample = buf.readUInt16LE(body + 14);
    } else if (id === "data") {
      if (bitsPerSample !== 16) throw new Error(`expected 16-bit PCM, got ${bitsPerSample}`);
      return { pcm: buf.subarray(body, body + size), sampleRate, channels };
    }
    offset = body + size + (size % 2);
  }
  throw new Error(`no data chunk in ${path}`);
}
```

### Task 2: Spike 1 — session continuity

Answers: does `gemini-3.5-live-translate-preview` accept `contextWindowCompression` and `sessionResumption`, does it emit `GoAway`, and how large is the seam on reconnect? The answer decides whether `SessionRotator` gets built at all.

- Modify: `docs/superpowers/spike-findings.md`

**Interfaces:**
- Consumes: `readWavAsPcm16`, `chunkPcm` from Task 1.
- Produces: a findings section. No code that later tasks import.

- [ ] **Step 1: Write the probe**

`spikes/01-session-continuity/run.ts`:
```ts
import { GoogleGenAI, Modality } from "@google/genai";
import { readWavAsPcm16, chunkPcm } from "../shared/pcm.ts";

const RUN_MS = 25 * 60 * 1000; // must exceed both the 10 min connection and 15 min session limits
const FRAME_BYTES = 640; // 20 ms @ 16 kHz mono s16le
const started = Date.now();

const log = (event: string, detail: Record<string, unknown> = {}) => {
  const t = ((Date.now() - started) / 1000).toFixed(1).padStart(7);
  console.log(`[${t}s] ${event} ${JSON.stringify(detail)}`);
};

const ai = new GoogleGenAI({});
const { pcm } = await readWavAsPcm16("spikes/sample-ko.wav");
const frames = chunkPcm(pcm, FRAME_BYTES);
log("loaded", { frames: frames.length, seconds: frames.length * 0.02 });

let handle: string | undefined;
let lastAudioAt = 0;
let reconnects = 0;

async function connect(resumeWith?: string) {
  return ai.live.connect({
    model: "gemini-3.5-live-translate-preview",
    config: {
      responseModalities: [Modality.AUDIO],
      inputAudioTranscription: {},
      outputAudioTranscription: {},
      translationConfig: { targetLanguageCode: "en", echoTargetLanguage: false },
      contextWindowCompression: { slidingWindow: {} },
      ...(resumeWith
        ? { sessionResumption: { handle: resumeWith } }
        : { sessionResumption: {} }),
    },
    callbacks: {
      onopen: () => log("open", { resumed: Boolean(resumeWith) }),
      onmessage: (m: any) => {
        if (m.sessionResumptionUpdate) {
          if (m.sessionResumptionUpdate.resumable && m.sessionResumptionUpdate.newHandle) {
            handle = m.sessionResumptionUpdate.newHandle;
            log("resumption-handle", { len: handle!.length });
          }
        }
        if (m.goAway) log("GOAWAY", { timeLeft: m.goAway.timeLeft });
        const audio = m.serverContent?.modelTurn?.parts?.find((p: any) => p.inlineData);
        if (audio) {
          const gap = lastAudioAt ? Date.now() - lastAudioAt : 0;
          if (gap > 2000) log("AUDIO-GAP", { gapMs: gap });
          lastAudioAt = Date.now();
        }
        const outText = m.serverContent?.outputTranscription?.text;
        if (outText) log("out", { text: outText.slice(0, 60) });
      },
      onerror: (e: any) => log("ERROR", { message: e?.message }),
      onclose: (e: any) => log("close", { reason: e?.reason, code: e?.code }),
    },
  });
}

let session = await connect();

// Feed audio in a loop until RUN_MS elapses, reconnecting when the socket dies.
let i = 0;
while (Date.now() - started < RUN_MS) {
  const frame = frames[i++ % frames.length];
  try {
    session.sendRealtimeInput({
      audio: { data: frame.toString("base64"), mimeType: "audio/pcm;rate=16000" },
    });
  } catch (err) {
    reconnects++;
    const gapStart = Date.now();
    log("RECONNECTING", { attempt: reconnects, withHandle: Boolean(handle) });
    session = await connect(handle);
    log("RECONNECTED", { seamMs: Date.now() - gapStart });
  }
  await new Promise((r) => setTimeout(r, 20));
}

log("done", { reconnects });
session.close();
process.exit(0);
```

`spikes/02-latency/run.ts`:

```ts
import { GoogleGenAI, Modality } from "@google/genai";
import { readWavAsPcm16, chunkPcm } from "../shared/pcm.ts";

const FRAME_BYTES = 640;
const ai = new GoogleGenAI({});
const { pcm } = await readWavAsPcm16("spikes/sample-ko.wav");
const frames = chunkPcm(pcm, FRAME_BYTES).slice(0, 250); // 5 s utterance

let sentAt = 0;
let firstAudioAt = 0;
let firstTranscriptAt = 0;

const session = await ai.live.connect({
  model: "gemini-3.5-live-translate-preview",
  config: {
    responseModalities: [Modality.AUDIO],
    inputAudioTranscription: {},
    outputAudioTranscription: {},
    translationConfig: { targetLanguageCode: "en", echoTargetLanguage: false },
    contextWindowCompression: { slidingWindow: {} },
  },
  callbacks: {
    onopen: () => console.log("open"),
    onmessage: (m: any) => {
      if (!firstTranscriptAt && m.serverContent?.outputTranscription?.text) {
        firstTranscriptAt = Date.now();
      }
      if (!firstAudioAt && m.serverContent?.modelTurn?.parts?.some((p: any) => p.inlineData)) {
        firstAudioAt = Date.now();
      }
    },
    onerror: (e: any) => console.error("error", e?.message),
    onclose: () => {},
  },
});

for (const frame of frames) {
  session.sendRealtimeInput({
    audio: { data: frame.toString("base64"), mimeType: "audio/pcm;rate=16000" },
  });
  await new Promise((r) => setTimeout(r, 20)); // real time, not a flood
}
sentAt = Date.now();
console.log("utterance sent");

await new Promise((r) => setTimeout(r, 15000));

console.log(JSON.stringify({
  audioLatencyMs: firstAudioAt ? firstAudioAt - sentAt : null,
  transcriptLatencyMs: firstTranscriptAt ? firstTranscriptAt - sentAt : null,
  transcriptLeadsAudioByMs:
    firstAudioAt && firstTranscriptAt ? firstAudioAt - firstTranscriptAt : null,
}, null, 2));

session.close();
process.exit(0);
```

- [ ] **Step 1: Write the streaming server**

ffmpeg produces a live WebM/Opus stream. The server buffers everything before
the first Cluster as the init segment, then relays clusters to all clients. A
late joiner receives init segment + live clusters.

`spikes/03-audio-playback/serve.ts`:
```ts
import { spawn } from "node:child_process";
import { createServer, type ServerResponse } from "node:http";
import { readFileSync } from "node:fs";

const CLUSTER_ID = Buffer.from([0x1f, 0x43, 0xb6, 0x75]);
const PORT = 8080;

const page = readFileSync(new URL("./index.html", import.meta.url));
const clients = new Set<ServerResponse>();

let initSegment: Buffer | null = null;
let pending = Buffer.alloc(0);

const ff = spawn("ffmpeg", [
  "-re",
  "-f", "lavfi",
  "-i", "sine=frequency=440:sample_rate=48000",
  "-c:a", "libopus",
  "-b:a", "24k",
  "-f", "webm",
  "-live", "1",
  "-cluster_time_limit", "200",
  "pipe:1",
]);
ff.stderr.on("data", () => {});

ff.stdout.on("data", (chunk: Buffer) => {
  if (initSegment === null) {
    pending = Buffer.concat([pending, chunk]);
    const at = pending.indexOf(CLUSTER_ID);
    if (at === -1) return;
    initSegment = pending.subarray(0, at);
    console.log(`init segment: ${initSegment.length} bytes`);
    const rest = pending.subarray(at);
    pending = Buffer.alloc(0);
    for (const res of clients) res.write(rest);
    return;
  }
  for (const res of clients) res.write(chunk);
});

createServer((req, res) => {
  if (req.url === "/" || req.url?.startsWith("/index.html")) {
    res.writeHead(200, { "content-type": "text/html; charset=utf-8" });
    res.end(page);
    return;
  }
  if (req.url?.startsWith("/stream.webm")) {
    if (!initSegment) {
      res.writeHead(503).end("stream not ready");
      return;
    }
    res.writeHead(200, {
      "content-type": "audio/webm",
      "cache-control": "no-store",
      "transfer-encoding": "chunked",
    });
    res.write(initSegment);
    clients.add(res);
    console.log(`client joined (${clients.size} total)`);
    req.on("close", () => {
      clients.delete(res);
      console.log(`client left (${clients.size} total)`);
    });
    return;
  }
  res.writeHead(404).end();
}).listen(PORT, "0.0.0.0", () => {
  console.log(`http://<your-lan-ip>:${PORT}`);
});
```
